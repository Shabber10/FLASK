## 1. What is CRUD in Plain English?

**CRUD** stands for the four basic actions performed on a database:
1.  **C**reate -> Insert new student records into MySQL (`INSERT INTO student`)
2.  **R**ead -> Query and fetch student records from MySQL (`SELECT FROM student`)
3.  **U**pdate -> Modify existing student records in MySQL (`UPDATE student`)
4.  **D**elete -> Remove student records from MySQL (`DELETE FROM student`)

---

## 2. Complete `mysql-connector-python` CRUD Implementation

### 1. Create (Insert New Student into MySQL)

To insert student data safely, **ALWAYS** use `%s` placeholders to prevent SQL Injection:

```python
@app.route('/add-student', methods=['POST'])
def add_student():
    sname = request.form.get('sname')
    sage = request.form.get('sage', type=int)
    smarks = request.form.get('smarks', type=float)
    scity = request.form.get('scity')

    conn = get_db_connection()
    cursor = conn.cursor()
    
    # SAFE: Use parameterized placeholders (%s)
    sql = "INSERT INTO student (sname, sage, smarks, scity) VALUES (%s, %s, %s, %s)"
    values = (sname, sage, smarks, scity)
    
    cursor.execute(sql, values)
    conn.commit() # MUST call commit() to write changes to MySQL disk!
    
    new_id = cursor.lastrowid # Retrieve auto-generated Primary Key ID
    
    cursor.close()
    conn.close()
    return f"Student added successfully with ID {new_id}!"
```

### 2. Read (Fetch Records from MySQL)

Use `cursor = conn.cursor(dictionary=True)` so fetched rows return as Python dictionaries:

```python
# A. Read ALL Student Records
@app.route('/students')
def list_students():
    conn = get_db_connection()
    cursor = conn.cursor(dictionary=True) # Returns dicts: {'id': 1, 'sname': 'Aarav', 'sage': 20, ...}
    
    cursor.execute("SELECT id, sname, sage, smarks, scity FROM student ORDER BY id DESC")
    students = cursor.fetchall()
    
    cursor.close()
    conn.close()
    return render_template('employees.html', students=students)

# B. Read SINGLE Student Record by ID
@app.route('/student/<int:std_id>')
def get_student(std_id):
    conn = get_db_connection()
    cursor = conn.cursor(dictionary=True)
    
    cursor.execute("SELECT * FROM student WHERE id = %s", (std_id,))
    student = cursor.fetchone() # Returns single dict or None
    
    cursor.close()
    conn.close()
    
    if not student:
        return jsonify({"error": "Student not found"}), 404
        
    return jsonify(student)
```

### 3. Update (Modify Existing MySQL Student Record)

```python
@app.route('/update-marks/<int:std_id>', methods=['POST'])
def update_marks(std_id):
    new_marks = request.form.get('smarks', type=float)
    
    conn = get_db_connection()
    cursor = conn.cursor()
    
    sql = "UPDATE student SET smarks = %s WHERE id = %s"
    cursor.execute(sql, (new_marks, std_id))
    conn.commit() # MUST call commit()!
    
    rows_affected = cursor.rowcount
    
    cursor.close()
    conn.close()
    
    if rows_affected == 0:
        return f"No student found with ID {std_id}", 404
        
    return f"Updated marks for student ID {std_id}!"
```

### 4. Delete (Remove MySQL Student Record)

```python
@app.route('/delete-student/<int:std_id>', methods=['POST'])
def delete_student(std_id):
    conn = get_db_connection()
    cursor = conn.cursor()
    
    sql = "DELETE FROM student WHERE id = %s"
    cursor.execute(sql, (std_id,))
    conn.commit() # MUST call commit()!
    
    cursor.close()
    conn.close()
    return f"Student ID {std_id} deleted successfully!"
```

## 3. SQL Injection Defense Rule 🛡️

> 🚨 **CRITICAL SECURITY RULE**: Never concatenate user input directly into SQL strings!

```python
# ❌ VULNERABLE TO SQL INJECTION (DO NOT DO THIS!):
user_city = request.args.get('city')
cursor.execute(f"SELECT * FROM student WHERE scity = '{user_city}'")

# ✅ 100% SAFE PARAMETERIZED QUERY (DO THIS!):
cursor.execute("SELECT * FROM student WHERE scity = %s", (user_city,))
```

## 4. Transaction Error Handling & Rollback (`conn.rollback()`)

Always wrap database operations in `try/except/finally` blocks to handle errors and roll back failed transactions safely:

```python
import mysql.connector

def create_student_safely(sname, sage, smarks, scity):
    conn = None
    try:
        conn = get_db_connection()
        cursor = conn.cursor()
        
        sql = "INSERT INTO student (sname, sage, smarks, scity) VALUES (%s, %s, %s, %s)"
        cursor.execute(sql, (sname, sage, smarks, scity))
        conn.commit()
        return True
    except mysql.connector.Error as err:
        if conn:
            conn.rollback() # Revert transaction on error!
        print(f"[MYSQL ERROR]: {err}")
        return False
    finally:
        if conn and conn.is_connected():
            cursor.close()
            conn.close()
```
