## 1. Database Table Schema (`student` Table)

```sql
CREATE DATABASE IF NOT EXISTS collegedb;
USE collegedb;

CREATE TABLE IF NOT EXISTS student (
    id INT AUTO_INCREMENT PRIMARY KEY,
    sname VARCHAR(100) NOT NULL,
    sage INT NOT NULL,
    smarks INT NOT NULL,
    scity VARCHAR(100) NOT NULL
);
```

## 2. Full Flask CRUD Route Implementations

```python
from flask import Flask, render_template, request, redirect
import mysql.connector

app = Flask(__name__)

# 1. Establish Database Connection
conn = mysql.connector.connect(
    host="localhost",
    user="root",
    password="your_password",
    database="collegedb"
)

# 2. Connection Check Route
@app.route('/')
def Home():
    return "Database connection is successful!"

# =============================================================================
# READ: Fetch All Student Records
# =============================================================================
@app.route('/getstudents', methods=['GET'])
def Getstudents():
    cursor = conn.cursor()
    cursor.execute("SELECT * FROM student")
    results = cursor.fetchall()
    cursor.close()
    return render_template('studentsdata.html', students=results)

# =============================================================================
# READ: Fetch Single Student Record by ID (Dictionary Cursor)
# =============================================================================
@app.route('/getstudentbyid/<int:sid>', methods=['GET'])
def Getstudentbyid(sid):
    cursor = conn.cursor(dictionary=True)
    cursor.execute("SELECT * FROM student WHERE id = %s", (sid,))
    result = cursor.fetchone()
    cursor.close()
    if not result:
        return {"message": f"Student ID {sid} not found"}, 404
    return result

# =============================================================================
# CREATE: Display Registration Form & Insert Student
# =============================================================================
@app.route('/register')
def Register():
    return render_template('register.html')

@app.route('/Addstudent', methods=['POST'])
def Addstudent():
    name = request.form['myname']
    age = request.form['myage']
    marks = request.form['mymarks']
    city = request.form['mycity']

    cursor = conn.cursor()
    query = "INSERT INTO student (sname, sage, smarks, scity) VALUES (%s, %s, %s, %s)"
    cursor.execute(query, (name, age, marks, city))
    conn.commit()  # Save changes permanently to database!
    cursor.close()
    return redirect('/getstudents')

# =============================================================================
# UPDATE: Fetch Student for Editing & Save Updates
# =============================================================================
@app.route('/editstudent/<int:sid>', methods=['GET'])
def Editstudent(sid):
    cursor = conn.cursor()
    cursor.execute("SELECT * FROM student WHERE id = %s", (sid,))
    result = cursor.fetchone()
    cursor.close()
    return render_template('editstudent.html', student=result)

@app.route('/updatestudent/<int:sid>', methods=['POST'])
def Updatestudent(sid):
    id = request.form['sid']
    name = request.form['sname']
    age = request.form['sage']
    marks = request.form['smarks']
    city = request.form['scity']

    cursor = conn.cursor()
    query = "UPDATE student SET sname = %s, sage = %s, smarks = %s, scity = %s WHERE id = %s"
    cursor.execute(query, (name, age, marks, city, id))
    conn.commit()
    cursor.close()
    return redirect('/getstudents')

# =============================================================================
# DELETE: Remove Student Record by ID
# =============================================================================
@app.route('/deletestudent/<int:sid>', methods=['GET'])
def Deletestudent(sid):
    cursor = conn.cursor()
    cursor.execute("DELETE FROM student WHERE id = %s", (sid,))
    conn.commit()
    cursor.close()
    return redirect('/getstudents')

if __name__ == '__main__':
    app.run(debug=True)
```
