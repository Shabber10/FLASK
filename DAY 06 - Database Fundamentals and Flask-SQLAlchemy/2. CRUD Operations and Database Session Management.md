## 1. What is CRUD in Plain English?

**CRUD** stands for the four basic actions performed on a database:
1.  **C**reate -> Insert new records into MySQL (`INSERT INTO`)
2.  **R**ead -> Query and fetch existing records from MySQL (`SELECT`)
3.  **U**pdate -> Modify existing records in MySQL (`UPDATE`)
4.  **D**elete -> Remove records from MySQL (`DELETE`)

---

## 2. Complete `mysql-connector-python` CRUD Implementation

### 1. Create (Insert New Record into MySQL)

To insert data safely, **ALWAYS** use `%s` placeholders to prevent SQL Injection:

```python
@app.route('/add-employee', methods=['POST'])
def add_employee():
    first_name = request.form.get('first_name')
    last_name = request.form.get('last_name')
    email = request.form.get('email')
    salary = request.form.get('salary', type=float)

    conn = get_db_connection()
    cursor = conn.cursor()
    
    # SAFE: Use parameterized placeholders (%s)
    sql = "INSERT INTO employees (first_name, last_name, email, salary) VALUES (%s, %s, %s, %s)"
    values = (first_name, last_name, email, salary)
    
    cursor.execute(sql, values)
    conn.commit() # MUST call commit() to write changes to MySQL disk!
    
    new_id = cursor.lastrowid # Retrieve auto-generated Primary Key ID
    
    cursor.close()
    conn.close()
    return f"Employee added successfully with ID {new_id}!"
```

---

### 2. Read (Fetch Records from MySQL)

Use `cursor = conn.cursor(dictionary=True)` so fetched rows return as Python dictionaries:

```python
# A. Read ALL Records
@app.route('/employees')
def list_employees():
    conn = get_db_connection()
    cursor = conn.cursor(dictionary=True) # Returns dicts: {'id': 1, 'first_name': 'Alice', ...}
    
    cursor.execute("SELECT id, first_name, last_name, email, salary FROM employees ORDER BY last_name")
    employees = cursor.fetchall()
    
    cursor.close()
    conn.close()
    return render_template('employees.html', employees=employees)

# B. Read SINGLE Record by ID
@app.route('/employee/<int:emp_id>')
def get_employee(emp_id):
    conn = get_db_connection()
    cursor = conn.cursor(dictionary=True)
    
    cursor.execute("SELECT * FROM employees WHERE id = %s", (emp_id,))
    employee = cursor.fetchone() # Returns single dict or None
    
    cursor.close()
    conn.close()
    
    if not employee:
        return jsonify({"error": "Employee not found"}), 404
        
    return jsonify(employee)
```

### 3. Update (Modify Existing MySQL Record)

```python
@app.route('/update-salary/<int:emp_id>', methods=['POST'])
def update_salary(emp_id):
    new_salary = request.form.get('salary', type=float)
    
    conn = get_db_connection()
    cursor = conn.cursor()
    
    sql = "UPDATE employees SET salary = %s WHERE id = %s"
    cursor.execute(sql, (new_salary, emp_id))
    conn.commit() # MUST call commit()!
    
    rows_affected = cursor.rowcount
    
    cursor.close()
    conn.close()
    
    if rows_affected == 0:
        return f"No employee found with ID {emp_id}", 404
        
    return f"Updated salary for employee ID {emp_id}!"
```

---

### 4. Delete (Remove MySQL Record)

```python
@app.route('/delete-employee/<int:emp_id>', methods=['POST'])
def delete_employee(emp_id):
    conn = get_db_connection()
    cursor = conn.cursor()
    
    sql = "DELETE FROM employees WHERE id = %s"
    cursor.execute(sql, (emp_id,))
    conn.commit() # MUST call commit()!
    
    cursor.close()
    conn.close()
    return f"Employee ID {emp_id} deleted successfully!"
```

## 3. SQL Injection Defense Rule 🛡️

> 🚨 **CRITICAL SECURITY RULE**: Never concatenate user input directly into SQL strings!

```python
# ❌ VULNERABLE TO SQL INJECTION (DO NOT DO THIS!):
user_email = request.args.get('email')
cursor.execute(f"SELECT * FROM employees WHERE email = '{user_email}'")

# ✅ 100% SAFE PARAMETERIZED QUERY (DO THIS!):
cursor.execute("SELECT * FROM employees WHERE email = %s", (user_email,))
```

## 4. Transaction Error Handling & Rollback (`conn.rollback()`)

Always wrap database operations in `try/except/finally` blocks to handle errors and roll back failed transactions safely:

```python
import mysql.connector

def create_employee_safely(first_name, last_name, email, salary):
    conn = None
    try:
        conn = get_db_connection()
        cursor = conn.cursor()
        
        sql = "INSERT INTO employees (first_name, last_name, email, salary) VALUES (%s, %s, %s, %s)"
        cursor.execute(sql, (first_name, last_name, email, salary))
        conn.commit()
        return True
    except mysql.connector.Error as err:
        if conn:
            conn.rollback() # Revert transaction on error!
        print(f"[MYSQL ERROR]: {err}")
        return False
    finally:
        if conn and conn.is_connected():
            cursor.close()
            conn.close()
```
